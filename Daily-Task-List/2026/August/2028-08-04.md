## 记录
1. PowerPoint Render管线的输入是生成好的List\[SlideContent\]，即PPT文件每页的具体内容，关键输出是SlidesRenderResult
```Python
# 关键输入
class SlideContent(BaseModel):
    index: int = Field(description="页序，1-based")
    title: str = Field(description="页标题")
    body: str = Field(default="", description="正文（markdown 要点）")
    notes: str | None = Field(default=None, description="演讲稿备注")
    layout: SlideLayout | None = Field(
        default=None, description="版式提示（帮 skill 选模板页）"
    )


# 关键输出
class Artifact(BaseModel):
    object_key: str = Field(default=..., description="对象存储中的键")
    name: str = Field(default=..., description="文件名")
    size_kb: int = Field(default=..., description="大小（KB）")


class SvgArtifact(Artifact):
    index: int = Field(default=..., description="页序，1-based")


class SlidesRenderResult(BaseModel):
    pptx: Artifact = Field(default=..., description="渲染出的 pptx")
    svgs: list[SvgArtifact] = Field(
        default_factory=list, description="逐页 SVG，按页序"
    )
    source_markdown: str = Field(default=..., description="实际渲染的 source.md")

```
