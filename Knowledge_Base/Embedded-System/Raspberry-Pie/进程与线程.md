## 方法一：使用线程 (Threading)

**基础知识：线程的概念**

- **进程 (Process):** 一个正在运行的程序的实例。每个进程都有自己独立的内存空间和系统资源。当你启动一个 Python 脚本时，操作系统会创建一个新的进程来执行它。
- **线程 (Thread):** 是进程内部的执行单元。一个进程可以包含多个线程。同一个进程内的线程共享进程的内存空间和系统资源，这使得线程间的通信比进程间通信更方便。
- **并发 (Concurrency):** 指的是在同一时间段内，多个任务看起来像是在同时执行。在单核 CPU 上，这是通过快速地在不同任务之间切换来实现的。
- **并行 (Parallelism):** 指的是在同一时刻，多个任务在不同的 CPU 核心上真正地同时执行。
- **多线程 (Multithreading):** 在一个进程中创建并运行多个线程，以实现并发或并行执行不同的任务。

**为什么线程适合串口监听？**

对于串口监听来说，我们希望程序在等待串口数据到来时不要阻塞主程序的其他操作（例如，处理摄像头画面、显示图像等）。使用一个独立的线程专门负责监听串口，可以让主线程继续执行其他任务。当串口线程接收到数据时，它可以将数据传递给主线程进行处理。

**Python 的 `threading` 模块**

Python 的 `threading` 模块提供了创建和管理线程的功能。一些常用的类和函数包括：

- `threading.Thread(target=function, args=(arg1, arg2, ...), daemon=True/False)`:

   创建一个新的线程对象。

  - `target`: 指定线程要执行的函数。
  - `args`: 以元组的形式传递给目标函数的参数。
  - `daemon`: 一个布尔值，指示线程是否为守护线程。守护线程在主线程退出时会自动结束。通常将后台任务（如串口监听）设置为守护线程。

- **`thread.start()`:** 启动线程，使其开始执行 `target` 函数。

- **`thread.join()`:** 等待线程执行结束。通常在主线程的最后调用，以确保所有子线程都完成后主线程才退出。

- **`threading.Lock()`:** 创建一个锁对象，用于同步多个线程对共享资源的访问，避免竞争条件。

- **`queue.Queue()`:** 提供线程安全的消息队列，用于在多个线程之间传递数据。

Python

```python
import cv2
import serial
import numpy as np
import pytesseract
import json
import threading
import time
from queue import Queue

class UART:
    def __init__(self, instance=None, baud_rate=9600):
        # 设置默认串口设备
        if instance is not None:
            self.instance = instance
        else:
            self.instance = "/dev/ttyS0"  # 树莓派4B的UART接口

        self.baud_rate = baud_rate

        # 初始化串口
        try:
            self.uart = serial.Serial(
                port=self.instance,
                baudrate=self.baud_rate,
                timeout=1  # 设置超时时间为1秒
            )
            if self.uart.is_open:
                print(f"串口已打开: {self.instance}, 波特率: {self.baud_rate}")
        except Exception as e:
            print(f"串口打开失败: {e}")
            self.uart = None

    def uart_transmit(self, data):
        if self.uart and self.uart.is_open:
            try:
                # 将数据编码为 JSON 字符串并发送
                json_data = json.dumps(data)  # 使用 json.dumps() 将 Python 对象转换为 JSON 字符串
                encoded_data = json_data.encode('utf-8')
                self.uart.write(encoded_data)
                print(f"发送数据: {json_data}")
                return len(encoded_data)  # 返回发送的字节数
            except Exception as e:
                print(f"发送数据失败: {e}")
        return 0

    def uart_receive(self, size=1024):
        if self.uart and self.uart.is_open:
            try:
                # 读取数据
                received_data = self.uart.read(size)
                if received_data:
                    # 解码接收到的数据
                    decoded_data = received_data.decode('utf-8', errors='ignore')
                    print(f"接收数据: {decoded_data}")
                    try:
                        # 尝试将接收到的数据解析为 JSON
                        return json.loads(decoded_data)  # 使用 json.loads() 将 JSON 字符串转换为 Python 对象
                    except json.JSONDecodeError:
                        # 如果不是 JSON，则返回原始字符串
                        return decoded_data
            except Exception as e:
                print(f"接收数据失败: {e}")
        return None

    def close(self):
        # 添加关闭串口的方法
        if self.uart and self.uart.is_open:
            self.uart.close()
            print(f"串口已关闭: {self.instance}")

class Camera:
    def __init__(self, camera_index=0, width=320, height=240, fourcc='YUYV', fps=30, show_width=960, show_height=720):
        self.camera_index = camera_index
        self.width = width
        self.height = height
        self.fourcc_str = fourcc
        self.fps = fps
        self.cap = None
        self.is_opened = False
        self.show_width = show_width
        self.show_height = show_height
        self._initialize_camera()

    def _initialize_camera(self):
        self.cap = cv2.VideoCapture(self.camera_index, cv2.CAP_V4L2)
        if not self.cap.isOpened():
            self.is_opened = False
            print("无法打开摄像头!")
        else:
            self.is_opened = True
            self.cap.set(cv2.CAP_PROP_FRAME_WIDTH, self.width)
            self.cap.set(cv2.CAP_PROP_FRAME_HEIGHT, self.height)
            self.cap.set(cv2.CAP_PROP_FOURCC, cv2.VideoWriter_fourcc(*self.fourcc_str))
            self.cap.set(cv2.CAP_PROP_FPS, self.fps)
            print(f"摄像头 {self.camera_index} 初始化完成，参数如下:")
            print(f"  分辨率: {self.width}x{self.height}")
            print(f"  像素格式: {self.fourcc_str}")
            print(f"  帧率: {self.fps}")

    def read_frame(self):
        if self.is_opened:
            ret, frame = self.cap.read()
            if not ret:
                print("无法读取帧!")
                return None
            return frame
        else:
            print("摄像头未打开!")
            return None

    def show_frame(self, window_name, frame):
        if frame is not None:
            cv2.namedWindow(window_name, cv2.WINDOW_NORMAL)
            cv2.resizeWindow(window_name, self.show_width, self.show_height)  # 修正了宽度和高度的顺序
            cv2.imshow(window_name, frame)

    def save_frame(self, filename, frame):
        if frame is not None:
            cv2.imwrite(filename, frame)

    def release(self):
        if self.is_opened and self.cap is not None:
            self.cap.release()
            print(f"摄像头 {self.camera_index} 资源已释放.")
            self.is_opened = False
            self.cap = None
        elif not self.is_opened:
            print(f"摄像头 {self.camera_index} 未打开，无需释放.")
        elif self.cap is None:
            print(f"摄像头捕获对象为空.")

    def destroy_all_windows(self):
        cv2.destroyAllWindows()

def uart_listener(uart_instance, data_queue):
    while True:
        data = uart_instance.uart_receive()
        if data is not None:
            data_queue.put(data)
        time.sleep(0.01) # 适当的延时，避免 CPU 占用过高



if __name__ == "__main__":
    camera = Camera(camera_index=0, width=320, height=240, fourcc='YUYV', fps=30)
    uart1 = UART(baud_rate=115200)
    use_cam = True
    data_queue = Queue()

    # 创建并启动串口监听线程
    uart_thread = threading.Thread(target=uart_listener, args=(uart1, data_queue), daemon=True)
    uart_thread.start()

    if camera.is_opened:
        while use_cam:
            # 检查数据队列是否有新的串口数据
            if not data_queue.empty():
                received_data = data_queue.get()
                print(f"主线程收到串口数据: {received_data}")
                # 在这里处理接收到的串口数据

            frame = camera.read_frame()
            if frame is not None:
                print("帧的形状:", frame.shape)
                shape = list(frame.shape)
                uart1.uart_transmit(shape)
                camera.show_frame("Camera", frame)
                camera.save_frame("test_frame.jpg", frame)
                if cv2.waitKey(1) & 0xFF == ord('q'):
                    break
            else:
                break
    else:
        print("摄像头初始化失败，程序退出.")

    camera.release()
    camera.destroy_all_windows()
    uart1.close() # 记得关闭串口
```

**代码解释：**

1. **`uart_listener(uart_instance, data_queue)` 函数:**
   - 这个函数是我们在独立线程中运行的目标函数。
   - 它在一个无限循环 `while True:` 中不断尝试从串口接收数据。
   - 如果 `uart_instance.uart_receive()` 返回了非 `None` 的数据，说明接收到了数据，我们使用 `data_queue.put(data)` 将数据放入 `data_queue` 中。`Queue` 是线程安全的，可以在多个线程之间安全地传递数据。
   - `time.sleep(0.01)`：为了避免这个监听线程过于频繁地检查串口，导致 CPU 占用过高，我们添加了一个短暂的延时。
2. **主线程 (`if __name__ == "__main__":`)**
   - 我们创建了一个 `Queue` 实例 `data_queue`，用于在串口监听线程和主线程之间传递接收到的数据。
   - 我们创建了一个 `threading.Thread` 对象 `uart_thread`：
     - `target=uart_listener`: 指定要执行的函数是 `uart_listener`。
     - `args=(uart1, data_queue)`: 将 `uart1` 实例和 `data_queue` 传递给 `uart_listener` 函数。
     - `daemon=True`: 将线程设置为守护线程。这意味着当主线程退出时，这个后台监听线程也会自动结束。
   - `uart_thread.start()`: 启动 `uart_thread`，使其开始执行 `uart_listener` 函数。
   - 主线程的 `while True:` 循环可以继续执行其他任务（在你的完整代码中，这里会是摄像头相关的处理）。
   - `if not data_queue.empty():`: 主线程定期检查 `data_queue` 是否为空。如果不为空，说明串口监听线程接收到了新的数据。
   - `received_data = data_queue.get()`: 使用 `data_queue.get()` 从队列中获取数据。这个操作是阻塞的，但因为我们只有在 `data_queue` 不为空时才调用 `get()`, 所以不会一直等待。
   - 在获取到数据后，你就可以在主线程中处理 `received_data` 了。
   - `time.sleep(0.1)`: 主线程的循环延时。
   - `try...except KeyboardInterrupt...finally`: 用于优雅地处理程序退出（例如按下 Ctrl+C），并在退出时关闭串口。





## 方法二：使用 `select` 模块

**基础知识：I/O 多路复用和 `select`**

- **I/O 操作:** 指的是程序与外部设备（例如文件、网络套接字、串口）之间的数据传输。
- **阻塞 (Blocking) I/O:** 当程序执行一个阻塞的 I/O 操作时，它会一直等待操作完成才能继续执行后续的代码。例如，`uart.read()` 默认是阻塞的，如果没有数据到来，程序会一直等待。
- **非阻塞 (Non-blocking) I/O:** 程序执行非阻塞的 I/O 操作时，如果操作不能立即完成，它会立即返回，而不会一直等待。
- **I/O 多路复用 (I/O Multiplexing):** 是一种允许程序同时监控多个 I/O 文件描述符的技术。通过这种机制，程序可以等待多个文件描述符中的任何一个变为“可读”或“可写”状态，然后只处理那些已经准备好的文件描述符，从而提高 I/O 效率。
- **`select` 模块:** Python 的 `select` 模块提供了 `select()` 函数，用于实现 I/O 多路复用。它可以监控文件描述符列表，等待它们中的一个或多个发生 I/O 事件。

**`select.select(rlist, wlist, xlist[, timeout])` 函数**

- **`rlist` (read list):** 需要监控的可读文件描述符列表。当列表中的任何一个文件描述符变为“可读”状态时，`select()` 会返回。
- **`wlist` (write list):** 需要监控的可写文件描述符列表。
- **`xlist` (exception list):** 需要监控的异常文件描述符列表。
- **`timeout` (可选):** 超时时间（秒）。如果指定了超时时间，`select()` 会在超时后返回，即使没有任何文件描述符就绪。如果 `timeout` 为 `None`，`select()` 会一直阻塞直到有文件描述符就绪。如果 `timeout` 为 `0`，`select()` 会立即返回，检查文件描述符的状态。
- **返回值:** `select()` 返回三个列表 `(readable, writable, exceptional)`，分别包含就绪的可读、可写和发生异常的文件描述符。

**为什么 `select` 适合串口监听？**

通过将串口的文件描述符添加到 `select()` 的 `rlist` 中，我们可以让程序等待，直到串口有数据到达（变为可读状态）时才进行读取操作。这样，在没有数据到达时，程序不会一直阻塞在 `uart.read()` 上，而是可以继续执行其他任务。

**代码讲解 (方法二的示例代码)**

```python
import select
import serial
import time

class UART:
    def __init__(self, instance=None, baud_rate=9600):
        # 设置默认串口设备
        if instance is not None:
            self.instance = instance
        else:
            self.instance = "/dev/ttyS0"  # 树莓派4B的UART接口

        self.baud_rate = baud_rate

        # 初始化串口
        try:
            self.uart = serial.Serial(
                port=self.instance,
                baudrate=self.baud_rate,
                timeout=1  # 设置超时时间为1秒
            )
            if self.uart.is_open:
                print(f"串口已打开: {self.instance}, 波特率: {self.baud_rate}")
        except Exception as e:
            print(f"串口打开失败: {e}")
            self.uart = None

    def uart_transmit(self, data):
        if self.uart and self.uart.is_open:
            try:
                # 将数据编码为 JSON 字符串并发送
                json_data = json.dumps(data)  # 使用 json.dumps() 将 Python 对象转换为 JSON 字符串

                self.uart.write(encoded_data)
                print(f"发送数据: {json_data}")
                return len(encoded_data)  # 返回发送的字节数
            except Exception as e:
                print(f"发送数据失败: {e}")
        return 0

    def uart_receive(self, size=1024):
        if self.uart and self.uart.is_open:
            try:
                # 读取数据
                received_data = self.uart.read(size)
                if received_data:
                    # 解码接收到的数据
                    decoded_data = received_data.decode('utf-8', errors='ignore')
                    print(f"接收数据: {decoded_data}")
                    try:
                        # 尝试将接收到的数据解析为 JSON
                        return json.loads(decoded_data)  # 使用 json.loads() 将 JSON 字符串转换为 Python 对象
                    except json.JSONDecodeError:
                        # 如果不是 JSON，则返回原始字符串
                        return decoded_data
            except Exception as e:
                print(f"接收数据失败: {e}")
        return None

    def close(self):
        # 添加关闭串口的方法
        if self.uart and self.uart.is_open:
            self.uart.close()
            print(f"串口已关闭: {self.instance}")

if __name__ == "__main__":
    uart1 = UART(baud_rate=115200)

    if uart1.uart is None:
        print("串口未成功初始化，程序退出.")
        exit()

    uart_fd = uart1.uart.fileno() # 获取串口的文件描述符

    try:
        while True:
            readable, _, _ = select.select([uart_fd], [], [], 0.1) # 监控串口是否可读，设置超时

            if readable:
                data = uart1.uart_receive() # 如果串口可读，则尝试读取数据
                if data is not None:
                    print(f"收到串口数据 (select): {data}")
                    # 在这里处理接收到的串口数据

            # 在这里执行你的其他任务（例如，摄像头处理）
            time.sleep(0.05) # 主循环的延时

    except KeyboardInterrupt:
        print("程序退出。")
    finally:
        uart1.close()
```

**代码解释：**

1. **获取文件描述符:**
   - `uart_fd = uart1.uart.fileno()`: 我们使用 `fileno()` 方法获取串口对象的文件描述符。文件描述符是操作系统用来标识打开的文件或 I/O 流的整数。`select()` 函数需要监控这些文件描述符。
2. **使用 `select.select()`:**
   - `readable, _, _ = select.select([uart_fd], [], [], 0.1)`:
     - `[uart_fd]`: 我们将包含串口文件描述符的列表传递给 `rlist`，表示我们想监控串口是否可读。
     - `[]`, `[]`: `wlist` 和 `xlist` 为空，因为我们不关心串口是否可写或发生异常。
     - `0.1`: 设置超时时间为 0.1 秒。`select()` 会等待最多 0.1 秒，看是否有文件描述符变为可读状态。如果超时时间内没有，它会返回空列表。
3. **检查可读状态:**
   - `if readable:`: 如果 `readable` 列表不为空，说明在超时时间内，`uart_fd` 对应的串口变为可读状态，即有数据到达。
4. **读取数据:**
   - `data = uart1.uart_receive()`: 我们调用 `uart1.uart_receive()` 来读取串口数据。注意，在 `UART` 类的初始化中，我将 `timeout` 设置为一个较小的值 (例如 `0.01`)，这样 `read()` 操作不会一直阻塞，即使在 `select()` 返回可读后，实际读取的数据可能不多或还没完全到达。
5. **执行其他任务:**
   - 在 `while` 循环中，即使没有接收到串口数据（`readable` 为空），程序也会继续执行循环中的其他任务（在你的完整代码中，这里会是摄像头相关的处理）。
   - `time.sleep(0.05)`: 主循环的延时。



**总结：**

- **线程方法**更直观，将串口监听放在一个独立的、并发执行的线程中，通过线程安全的消息队列与主线程通信。适合需要持续在后台监听并处理数据的场景。
- **`select` 方法**利用了操作系统提供的 I/O 多路复用机制，允许程序同时监控多个 I/O 流的状态，并在它们就绪时进行处理。更底层，可能在某些高并发 I/O 场景下更高效，但理解和实现可能稍微复杂一些。
